def makeCreditMultiplierSafe(battleClass):

    def getThrowTarget(self, battle):
        #Prefer the highest-level Cog we can kill this turn.
            throwTarget = None
            lowestLvlSuit = None
            throwDmg = ToontownBattleGlobals.getAvPropDamage(4, cjMaxer.restock.desiredLevel3, base.localAvatar.experience.getExp(4), base.localAvatar.checkGagBonus(4, cjMaxer.restock.desiredLevel3))
            for suit in battle.suits:
                suitLvl = suit.getActualLevel()
                if lowestLvlSuit is None or lowestLvlSuit.getActualLevel() > suitLvl:
                    lowestLvlSuit = suit

                hasKb = False
                try:
                    hasKb = battle.suits.index(suit) == battle.activeSuits.index(suit) and battle.isSuitLured(suit)
                except ValueError:
                    hasKb = False

                if hasKb and (throwDmg + (throwDmg * 0.5)) >= suit.getHP():
                    if throwTarget is None or throwTarget.getActualLevel() < suitLvl:
                        throwTarget = suit

                elif throwDmg >= suit.getHP():
                    if throwTarget is None or throwTarget.getActualLevel() < suitLvl:
                        throwTarget = suit

            if throwTarget is None:
                #Can't kill anything: hit the lowest-level Cog for the quickest kill.
                throwTarget = lowestLvlSuit
                if throwTarget is None:
                    throwTarget = battle.suits[0]

            return throwTarget

    def attack(self):
        try:
            for battle in base.cr.doFindAll('battle'):
                attackSuit = self.getThrowTarget(battle)
                wantThrow = True
                numUnlured = 0
                for suit in battle.suits:
                    if not battle.isSuitLured(suit):
                        numUnlured += 1

                if numUnlured > 1 or (self.isHealing and numUnlured != 0):
                    if base.localAvatar.inventory.inventory[cjMaxer.restock.firstTrack][cjMaxer.restock.desiredLevel1]>0:
                        battle.sendUpdate('requestAttack', [cjMaxer.restock.firstTrack, cjMaxer.restock.desiredLevel1, battle.suits[0].doId])
                        wantThrow = False
                    else:
                        break

                elif (base.localAvatar.getHp()<(base.localAvatar.getMaxHp()*0.9) or self.isHealing) and numUnlured == 0:
                    if base.localAvatar.inventory.inventory[cjMaxer.restock.secondTrack][cjMaxer.restock.desiredLevel2]>0:
                        battle.sendUpdate('requestAttack', [cjMaxer.restock.secondTrack, cjMaxer.restock.desiredLevel2, base.localAvatar.doId])
                        wantThrow = self.isHealing
                    else:
                        break

                if wantThrow:
                    if base.localAvatar.inventory.inventory[cjMaxer.restock.thirdTrack][cjMaxer.restock.desiredLevel3]>0:
                        battle.sendUpdate('requestAttack', [cjMaxer.restock.thirdTrack, cjMaxer.restock.desiredLevel3, attackSuit.doId])
                    else:
                        break
        except:
            pass

    def enterRandomBattle(self):
        battles = []
        cogList = base.cr.doFindAll("render")
        for x in cogList:
            if isinstance(x, DistributedSuit):
                if x.activeState == 6:
                    battles.append(x)
        if battles != []:
            self.walk()
            battle = random.choice(battles)
            pos, hpr = battle.getPos(), battle.getHpr()
            base.localAvatar.setPosHpr(pos, hpr)
            battle.d_requestBattle(pos, hpr)
            self.lastCogId=battle.doId
        else:
            pass

    def useResistanceUnite(self,menuType):
        #Use one owned unite of menuType on ourselves.
        try:
            from toontown.chat import ResistanceChat
            msgs=base.localAvatar.resistanceMessages
        except:
            return False
        for pair in msgs:
            try:
                textId,charges=pair[0],pair[1]
                matches=charges>0 and ResistanceChat.decodeId(textId)[0]==menuType
            except:
                continue
            if not matches:
                continue
            try:
                base.localAvatar.sendUpdate('reqSCResistance',[textId,[base.localAvatar.doId]])
                return True
            except:
                return False
        return False

        for shard in base.cr.activeDistrictMap:
            if base.cr.activeDistrictMap[shard].avatarCount<30:
                emptyShardList.append(shard)
        if emptyShardList==[]:
            emptyShardList=[650000000, 754000000, 608000000, 756000000, 658000000, 712000000, 360000000, 410000000, 620000000, 454000000, 726000000, 362000000, 688000000, 316000000]
        if base.localAvatar.defaultShard not in emptyShardList:
            cjMaxer.autoerGui.display.newLine('TPing to an empty district')
            base.cr.playGame.getPlace().requestTeleport(13000,13000,random.choice(emptyShardList),None)
        else:
            self.checkNotices()
        cjMaxer.restock.noToons.loop()
        self.stallCount=0
        self.lastSignature=None
        self.watchdogToken=getattr(self,'watchdogToken',0)+1
        self.stallWatchdog(self.watchdogToken)
        #Arm a 15-min district hop; applied between runs, never mid-run.
        self.wantDistrictHop=False
        self.districtHopToken=getattr(self,'districtHopToken',0)+1
        self.scheduleDistrictHop(self.districtHopToken)

    def scheduleDistrictHop(self,token):
        Sequence(Wait(900),Func(self.fireDistrictHop,token)).start()

    def fireDistrictHop(self,token):
        #Only the newest timer keeps looping.
        if token!=getattr(self,'districtHopToken',0) or not self.shouldContinue:
            return
        self.wantDistrictHop=True
        self.scheduleDistrictHop(token)

    def hopToFreshDistrict(self):
        #Hop to an empty Lawbot HQ district. True if a hop started.
        self.wantDistrictHop=False
        cur=base.localAvatar.defaultShard
        empties=[]
        try:
            for shard in base.cr.activeDistrictMap:
                if shard==cur:
                    continue
                if base.cr.activeDistrictMap[shard].avatarCount<30:
                    empties.append(shard)
        except:
            pass
        if empties==[]:
            return False
        cjMaxer.autoerGui.display.newLine('Refreshing district (15 min)')
        self.walk()
        try:
            base.cr.playGame.getPlace().requestTeleport(13000,13000,random.choice(empties),None)
            return True
        except:
            return False

    def progressSignature(self):
        #Progress snapshot - HP/notices/floorNum all freeze when stuck.
        floorNum=getattr(cjMaxer.daAutoer,'floorNum',-1)
        try:
            return (base.localAvatar.getZoneId(), base.localAvatar.getHp(),
                    base.localAvatar.getTotalMoney(), self.bossCount,
                    self.getNoticesLeft(), floorNum)
        except:
            return None

    def stallWatchdog(self,token):
        #Only the newest watchdog keeps looping.
        if token!=getattr(self,'watchdogToken',0) or not self.shouldContinue:
            return
        if cjMaxer.jellybeanFarmer.farming:
            #Fishing looks static between catches - don't police it.
            self.stallCount=0
            self.lastSignature=None
        else:
            sig=self.progressSignature()
            if sig is not None and sig==self.lastSignature:
                self.stallCount+=1
            else:
                self.stallCount=0
                self.lastSignature=sig
            #About 2 min of zero progress means we're stuck.
            if self.stallCount>=8:
                self.stallCount=0
                self.lastSignature=None
                cjMaxer.autoerGui.display.newLine('Stalled, restarting run')
                self.forceRestart()
        Sequence(Wait(15),Func(self.stallWatchdog,token)).start()

    def forceRestart(self):
        #Dead end - reset and restart at Lawbot HQ.
        cjMaxer.restock.wasInBattle=False
        self.isHealing=False
        self.isGlitched=False
        try:
            cjMaxer.daAutoer.attackSeq.finish()
            cjMaxer.daAutoer.gainLaffSeq.finish()
            cjMaxer.daAutoer.generateAgain()
        except:
            pass
        try:
            base.localAvatar.collisionsOn()
        except:
            pass
        self.walk()
        self.teleBack()

    def revertFunctions(self):
        LocalToon.LocalToon.setAnimState=self.oldSetAnim
        BRPlayground.BRPlayground.__init__=self.oldBrrInit
        LawbotHQExterior.LawbotHQExterior.enterWalk=self.oldEnterWalk
        DistributedBattle.DistributedBattle.enterReward=self.oldEnterReward
        DistributedSuit.denyBattle=self.oldDenyBattle
        DistributedObject.DistributedObject.sendUpdate=self.oldSendUpdate
        HealthForceAcknowledge.HealthForceAcknowledge.enter=self.oldEnterHealth
        DistributedBoardingParty.DistributedBoardingParty.postInvite=self.oldPostInvite

class JellybeanFarmer:
    DREAMLAND_ZONE=9000
    LOW_JELLYBEANS=2000
    BANK_FULL=12000

    def __init__(self,maxer):
        self.maxer=maxer
        self.farming=False
        self._noToonsRunning=False

    def inCurrentShard(self,do):
        #Old-shard objects linger after a hop; a doId floors to its district id.
        try:
            return (do.doId-(do.doId%1000000))==base.localAvatar.defaultShard
        except:
            return False

    def firstInShard(self,className):
        for do in base.cr.doFindAll(className):
            if self.inCurrentShard(do):
                return do
        return None

    def spotsInShard(self):
        spots=[]
        for s in base.cr.doFindAll('DistributedFishingSpot'):
            if self.inCurrentShard(s):
                spots.append(s)
        return spots

    def shouldFarm(self):
        try:
            if base.localAvatar.cogLevels[0]==49:
                return False
            return base.localAvatar.getTotalMoney()<self.LOW_JELLYBEANS
        except:
            return False

    def startFarming(self):
        if self.farming:
            return
        self.farming=True
        self.maxer.autoerGui.display.newLine('Low on jellybeans')
        try:
            self._noToonsRunning=self.maxer.restock.noToons.isPlaying()
            self.maxer.restock.noToons.finish()
        except:
            self._noToonsRunning=False
        #Beans from unites first; fish only if we have none.
        if self.tryBeanUnite():
            return
        self.beginFishing()

    def tryBeanUnite(self):
        #Use one Money unite, then check the total after it lands. True if used.
        if not self.farming:
            return False
        if self.maxer.otherFunctions.useMoneyUnite():
            self.maxer.autoerGui.display.newLine('Using jellybean unite')
            Sequence(Wait(1.5),Func(self.afterBeanUnite)).start()
            return True
        return False

    def afterBeanUnite(self):
        if not self.farming:
            return
        try:
            total=base.localAvatar.getTotalMoney()
        except:
            total=0
        self.maxer.autoerGui.display.newLine('Jellybeans: '+str(total))
        if total>=self.BANK_FULL:
            self.stopFarming()
            return
        #Still low: another unite, else fish.
        if not self.tryBeanUnite():
            self.beginFishing()

    def beginFishing(self):
        if not self.farming:
            return
        self.maxer.autoerGui.display.newLine('Fishing at Dreamland')
        self.maxer.otherFunctions.walk()
        Sequence(Wait(1),Func(self.goDreamland)).start()

    def goDreamland(self):
        try:
            self.maxer.otherFunctions.walk()
            base.cr.playGame.getPlace().handleBookCloseTeleport(self.DREAMLAND_ZONE,self.DREAMLAND_ZONE)
        except:
            pass
        Sequence(Wait(4),Func(self.enterPond)).start()

    def enterPond(self):
        if not self.farming:
            return
        #No Fisherman means the NPCs unloaded - hop rather than fish a dead end.
        if not self.firstInShard('Fisherman'):
            self.noFisherTries=getattr(self,'noFisherTries',0)+1
            if self.noFisherTries>=3:
                self.noFisherTries=0
                self.hopDistrict()
            else:
                Sequence(Wait(2),Func(self.enterPond)).start()
            return
        self.noFisherTries=0
        #Stale spots never spawn fish.
        spots=self.spotsInShard()
        if spots==[]:
            Sequence(Wait(2),Func(self.enterPond)).start()
            return
        try:
            spots[0].sendUpdate('requestEnter',[])
        except:
            pass
        self.maxer.autoerGui.display.newLine('Fishing for jellybeans')
        Sequence(Wait(2),Func(self.catchFish)).start()

    def hopDistrict(self):
        #Hop districts to refresh the unloaded NPCs so the Fisherman reappears.
        if not self.farming:
            return
        self.maxer.autoerGui.display.newLine('No fisherman here, switching district')
        cur=base.localAvatar.defaultShard
        others=[]
        populated=[]
        try:
            for shard in base.cr.activeDistrictMap:
                if shard==cur:
                    continue
                others.append(shard)
                if base.cr.activeDistrictMap[shard].avatarCount>0:
                    populated.append(shard)
        except:
            pass
        if populated:
            pool=populated
        else:
            pool=others
        try:
            self.maxer.otherFunctions.walk()
        except:
            pass
        if pool:
            try:
                base.cr.playGame.getPlace().requestTeleport(self.DREAMLAND_ZONE,self.DREAMLAND_ZONE,random.choice(pool),None)
            except:
                pass
        Sequence(Wait(5),Func(self.enterPond)).start()

    def catch(self):
        try:
            base.localAvatar.wakeUp()
            fish=None
            for t in base.cr.doFindAll('FishingTarget'):
                if self.inCurrentShard(t):
                    fish=t
                    break
            if fish is None:
                return
            for fp in base.cr.doFindAll('FishingPond'):
                if self.inCurrentShard(fp):
                    fp.d_hitTarget(fish)
        except:
            pass

    def catchFish(self):
        if not self.farming:
            return
        seq=Sequence()
        for i in range(23):
            seq.append(Func(self.catch))
            seq.append(Wait(0.05))
        seq.append(Wait(1))
        seq.append(Func(self.sell))
        seq.start()

    def sell(self):
        if not self.farming:
            return
        fisherman=self.firstInShard('Fisherman')
        if not fisherman:
            self.hopDistrict()
            return
        try:
            fisherman.sendUpdate('avatarEnter')
            fisherman.sendUpdate('completeSale',[1])
        except:
            pass
        Sequence(Wait(1),Func(self.afterSell)).start()

    def afterSell(self):
        if not self.farming:
            return
        try:
            wallet=base.localAvatar.getMoney()
            if wallet>0:
                base.cr.bankManager.d_transferMoney(wallet)
        except:
            pass
        try:
            total=base.localAvatar.getTotalMoney()
        except:
            total=0
        self.maxer.autoerGui.display.newLine('Jellybeans: '+str(total))
        if total>=self.BANK_FULL:
            self.stopFarming()
        else:
            self.catchFish()

    def stopFarming(self):
        self.farming=False
        self.maxer.autoerGui.display.newLine('Bank full - resuming')
        try:
            if self._noToonsRunning:
                self.maxer.restock.noToons.loop()
        except:
            pass
        self.maxer.otherFunctions.walk()
        #Return to an empty district so runs stay solo.
        Sequence(Wait(1),Func(self.returnToEmptyDistrict)).start()

    def returnToEmptyDistrict(self):
        empties=[]
        try:
            for shard in base.cr.activeDistrictMap:
                if base.cr.activeDistrictMap[shard].avatarCount<30:
                    empties.append(shard)
        except:
            pass
        try:
            self.maxer.otherFunctions.walk()
        except:
            pass
        try:
            if empties:
                base.cr.playGame.getPlace().requestTeleport(13000,13000,random.choice(empties),None)
            else:
                self.maxer.otherFunctions.teleBack()
        except:
            self.maxer.otherFunctions.teleBack()
