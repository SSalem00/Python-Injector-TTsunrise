from direct.interval.IntervalGlobal import *
from toontown.estate import GardenGlobals
from toontown.estate import DistributedGardenPlot
from toontown.estate import DistributedFlower
from toontown.estate import DistributedGagTree
from toontown.estate import DistributedLawnDecor
from toontown.estate import DistributedPlantBase
from toontown.estate import DistributedEstate
from toontown.toonbase import ToontownGlobals
from toontown.hood import ZoneUtil


class GardenMaxer:

    DAY_SECONDS = 24 * 60 * 60

    def __init__(self):
        self.shouldContinue = False
        self.dailyLoop = False
        self.plantNew = True
        self.sellAll = True
        self.harvested = 0
        self.watered = 0
        self.planted = 0
        self.sold = 0
        self.loopTask = None
        self.installSafety()
        self.autoerGui = AutoerGui('Garden Maxer',
                                   ['Run Time', 'Varieties Owned', 'Varieties Left', 'Harvested', 'Planted'],
                                   ['self.formattedTime',
                                    'str(gardenMaxer.ownedCount())',
                                    'str(gardenMaxer.missingCount())',
                                    'str(gardenMaxer.harvested)',
                                    'str(gardenMaxer.planted)'],
                                   'GARDEN MAXER', None)

    # Stub out setMovie - the real one crashes the client when we fire updates directly.
    SAFE_CLASSES = (DistributedLawnDecor.DistributedLawnDecor,
                    DistributedPlantBase.DistributedPlantBase,
                    DistributedGardenPlot.DistributedGardenPlot,
                    DistributedFlower.DistributedFlower,
                    DistributedGagTree.DistributedGagTree)

    def installSafety(self):
        for cls in self.SAFE_CLASSES:
            try:
                if '_gmOrigSetMovie' not in cls.__dict__:
                    cls._gmOrigSetMovie = cls.__dict__.get('setMovie', None)

                    def _gmSetMovie(decSelf, mode, avId):
                        try:
                            if avId == base.localAvatar.doId:
                                decSelf.movieDone()
                        except:
                            pass
                    cls.setMovie = _gmSetMovie
            except:
                pass

    def removeSafety(self):
        for cls in self.SAFE_CLASSES:
            try:
                if '_gmOrigSetMovie' in cls.__dict__:
                    orig = cls._gmOrigSetMovie
                    if orig is None:
                        try:
                            del cls.setMovie
                        except:
                            pass
                    else:
                        cls.setMovie = orig
                    del cls._gmOrigSetMovie
            except:
                pass

    # Every normal flower; special != -1 is a gag tree / statuary.
    def allVarieties(self):
        out = []
        try:
            for species in GardenGlobals.getFlowerSpecies():
                varieties = GardenGlobals.getFlowerVarieties(species)
                for v in range(len(varieties)):
                    recipeKey = varieties[v][0]
                    recipe = GardenGlobals.Recipes.get(recipeKey)
                    if recipe is None:
                        continue
                    if recipe.get('special', -1) != -1:
                        continue
                    out.append((species, v))
        except:
            pass
        return out

    def beansFor(self, species, variety):
        try:
            recipeKey = GardenGlobals.getFlowerVarieties(species)[variety][0]
            return GardenGlobals.Recipes[recipeKey]['beans']
        except:
            return ''

    def costFor(self, species, variety):
        return len(self.beansFor(species, variety))

    def isOwned(self, species, variety):
        try:
            return base.localAvatar.flowerCollection.hasFlower(species, variety)
        except:
            return 0

    def flowerName(self, species, variety):
        try:
            from toontown.toonbase import TTLocalizer
            return TTLocalizer.FlowerSpeciesNames[species]
        except:
            return 'S%s V%s' % (species, variety)

    def ownedCount(self):
        n = 0
        for f in self.allVarieties():
            if self.isOwned(f[0], f[1]):
                n += 1
        return n

    def missingCount(self):
        return len(self.allVarieties()) - self.ownedCount()

    # Shovel's bean cap - the server rejects longer recipes.
    def maxBeans(self):
        try:
            shovel = getattr(base.localAvatar, 'shovel', 0)
            skill = getattr(base.localAvatar, 'shovelSkill', 0)
            return GardenGlobals.getShovelPower(shovel, skill)
        except:
            return 1

    # Uncollected varieties we can afford, cheapest first.
    def missingCandidates(self):
        maxB = self.maxBeans()
        out = []
        for (s, v) in self.allVarieties():
            if self.isOwned(s, v):
                continue
            c = self.costFor(s, v)
            if c <= 0 or c > maxB:
                continue
            out.append((c, s, v))
        out.sort()
        return out

    # Owned recipes to replant for shovel skill, cap-cost first.
    def grindCandidates(self):
        maxB = self.maxBeans()
        best = []
        rest = []
        for (s, v) in self.allVarieties():
            c = self.costFor(s, v)
            if c <= 0 or c > maxB:
                continue
            if c == maxB:
                best.append((c, s, v))
            else:
                rest.append((c, s, v))
        rest.sort()
        rest.reverse()
        return best + rest

    def getMoney(self):
        try:
            return base.localAvatar.getMoney()
        except:
            return 0

    def inEstate(self):
        # Estate zones are dynamic, so check the Place, not the zone id.
        try:
            if 'Estate' in base.cr.playGame.getPlace().__class__.__name__:
                return True
        except:
            pass
        try:
            if 'Estate' in base.cr.playGame.hood.__class__.__name__:
                return True
        except:
            pass
        try:
            if ZoneUtil.isEstate(base.localAvatar.getZoneId()):
                return True
        except:
            pass
        try:
            if ZoneUtil.getHoodId(base.localAvatar.getZoneId()) == ToontownGlobals.MyEstate:
                return True
        except:
            pass
        return False

    # Empty flower plots (planted ones become DistributedFlower).
    def myFlowerPlots(self):
        out = []
        try:
            plots = base.cr.doFindAllInstances(DistributedGardenPlot.DistributedGardenPlot)
        except:
            plots = []
        for p in plots:
            try:
                if getattr(p, 'plotType', None) != GardenGlobals.FLOWER_TYPE:
                    continue
                if not p.canBePlanted():
                    continue
                out.append(p)
            except:
                pass
        return out

    def myFlowers(self):
        out = []
        try:
            flowers = base.cr.doFindAllInstances(DistributedFlower.DistributedFlower)
        except:
            flowers = []
        for f in flowers:
            try:
                if f.getOwnerId() == base.localAvatar.doId:
                    out.append(f)
            except:
                out.append(f)
        return out

    # Trees are watered only, never harvested.
    def myTrees(self):
        out = []
        try:
            trees = base.cr.doFindAllInstances(DistributedGagTree.DistributedGagTree)
        except:
            trees = []
        for t in trees:
            try:
                if t.getOwnerId() == base.localAvatar.doId:
                    out.append(t)
            except:
                out.append(t)
        return out

    def isReady(self, flower):
        try:
            return flower.isGTEFullGrown()
        except:
            try:
                return flower.growthLevel >= flower.growthThresholds[1]
            except:
                return False

    def needsWater(self, flower):
        try:
            return flower.waterLevel < flower.maxWaterLevel
        except:
            return True

    def start(self):
        self.shouldContinue = True
        self.harvested = 0
        self.watered = 0
        self.planted = 0
        self.sold = 0
        self.autoerGui.startTimer()
        if not self.inEstate():
            try:
                zid = base.localAvatar.getZoneId()
                pname = base.cr.playGame.getPlace().__class__.__name__
                self.autoerGui.display.newLine('Not estate? zone=%s place=%s' % (zid, pname))
            except:
                pass
            self.autoerGui.display.newLine('Go to YOUR ESTATE first, then Start')
            base.localAvatar.setSystemMessage(0, 'Garden Maxer: stand in your own estate before starting.')
            self.setStop()
            return
        try:
            started = base.localAvatar.getGardenStarted()
        except:
            started = True
        if not started:
            self.autoerGui.display.newLine('Start gardening at Toon HQ first (get a shovel)')
            base.localAvatar.setSystemMessage(0, 'Garden Maxer: get your shovel from Toon HQ before gardening.')
            self.setStop()
            return
        self.walk()
        Sequence(Wait(1), Func(self.doPass)).start()

    # Waits let each sendUpdate settle; watering runs last to catch new plants.
    def doPass(self):
        if not self.shouldContinue:
            return
        self.autoerGui.display.newLine('--- Daily garden pass ---')
        seq = Sequence()
        seq.append(Func(self.sellBasket))
        seq.append(Wait(1.5))
        for f in self.myFlowers():
            if self.isReady(f):
                seq.append(Func(self.harvest, f))
                seq.append(Wait(1.0))
        seq.append(Func(self.sellBasket))
        seq.append(Wait(1.5))
        seq.append(Wait(2))
        seq.append(Func(self.plantEmpties))
        seq.append(Wait(3))
        seq.append(Func(self.waterAll))
        seq.start()

    def waterAll(self):
        if not self.shouldContinue:
            return
        wseq = Sequence()
        for f in self.myFlowers():
            if not self.isReady(f) and self.needsWater(f):
                wseq.append(Func(self.water, f))
                wseq.append(Wait(1.0))
        for t in self.myTrees():
            if self.needsWater(t):
                wseq.append(Func(self.water, t))
                wseq.append(Wait(1.0))
        wseq.append(Func(self.finishPass))
        wseq.start()

    def harvest(self, flower):
        if not self.shouldContinue:
            return
        try:
            base.localAvatar.setPos(flower.getPos(render))
        except:
            pass
        try:
            flower.sendUpdate('removeItem', [])
            self.harvested += 1
            self.autoerGui.display.newLine('Harvested ' + self.flowerName(flower.getSpecies(), flower.getVariety()))
        except:
            pass

    def water(self, flower):
        if not self.shouldContinue:
            return
        try:
            base.localAvatar.setPos(flower.getPos(render))
        except:
            pass
        try:
            flower.sendUpdate('waterPlant')
            self.watered += 1
        except:
            pass
        # Some servers only commit water on the done update.
        try:
            flower.sendUpdate('waterPlantDone')
        except:
            pass

    # Cash in the whole basket.
    def sellBasket(self):
        if not self.shouldContinue or not self.sellAll:
            return
        try:
            count = len(base.localAvatar.flowerBasket.flowerList)
        except:
            count = 0
        if count <= 0:
            return
        estate = self.getEstate()
        if estate is None:
            self.autoerGui.display.newLine('No estate DO found - cannot sell')
            return
        try:
            estate.sendUpdate('completeFlowerSale', [1])
            self.sold += count
            self.autoerGui.display.newLine('Sold %s flowers from the basket' % count)
        except:
            pass

    def getEstate(self):
        try:
            ests = base.cr.doFindAllInstances(DistributedEstate.DistributedEstate)
            if ests:
                return ests[0]
        except:
            pass
        try:
            return base.cr.doFind('Estate')
        except:
            return None

    def plantEmpties(self):
        if not self.shouldContinue or not self.plantNew:
            return
        candidates = self.missingCandidates()
        grind = None
        gidx = 0
        money = self.getMoney()
        idx = 0
        for plot in self.myFlowerPlots():
            # Skip what we can't afford.
            while idx < len(candidates) and candidates[idx][0] > money:
                idx += 1
            if idx < len(candidates):
                c, s, v = candidates[idx]
                idx += 1
            else:
                if grind is None:
                    grind = self.grindCandidates()
                    self.autoerGui.display.newLine('No new varieties affordable - grinding %s-bean recipes for shovel skill' % self.maxBeans())
                pick = None
                for n in range(len(grind)):
                    cand = grind[(gidx + n) % len(grind)]
                    if cand[0] <= money:
                        pick = cand
                        gidx = (gidx + n + 1) % len(grind)
                        break
                if pick is None:
                    self.autoerGui.display.newLine('Not enough jellybeans to plant (%s left)' % money)
                    break
                c, s, v = pick
            money -= c
            try:
                base.localAvatar.setPos(plot.getPos(render))
            except:
                pass
            try:
                plot.sendUpdate('plantFlower', [s, v])
                self.planted += 1
                self.autoerGui.display.newLine('Planted ' + self.flowerName(s, v) +
                                               ' (' + self.beansFor(s, v) + ')')
            except:
                pass

    def finishPass(self):
        if not self.shouldContinue:
            return
        self.autoerGui.display.newLine('Pass done: harvested %s, watered %s, sold %s, planted %s'
                                       % (self.harvested, self.watered, self.sold, self.planted))
        base.localAvatar.setSystemMessage(0, 'Garden Maxer: pass done - %s harvested, %s watered, %s sold, %s planted.'
                                          % (self.harvested, self.watered, self.sold, self.planted))
        if self.dailyLoop:
            self.autoerGui.display.newLine('Daily loop ON - next pass in ~24h')
            self.loopTask = Sequence(Wait(self.DAY_SECONDS), Func(self.repeat))
            self.loopTask.start()
        else:
            self.setStop()

    def repeat(self):
        if not self.shouldContinue:
            return
        self.harvested = 0
        self.planted = 0
        self.watered = 0
        self.sold = 0
        if self.inEstate():
            self.doPass()
        else:
            self.autoerGui.display.newLine('Not in estate at loop time - skipping, retry in 1h')
            self.loopTask = Sequence(Wait(3600), Func(self.repeat))
            self.loopTask.start()

    def walk(self):
        try:
            base.cr.playGame.getPlace().fsm.forceTransition('walk')
        except:
            pass

    def loopOn(self):
        self.dailyLoop = True
        base.localAvatar.setSystemMessage(0, 'Garden Maxer: Daily loop ON (re-runs every 24h while open)')
        self.autoerGui.display.newLine('Daily loop ON')

    def loopOff(self):
        self.dailyLoop = False
        base.localAvatar.setSystemMessage(0, 'Garden Maxer: Daily loop OFF (single pass)')
        self.autoerGui.display.newLine('Daily loop OFF')

    def plantOn(self):
        self.plantNew = True
        self.autoerGui.display.newLine('Replanting ON')

    def plantOff(self):
        self.plantNew = False
        self.autoerGui.display.newLine('Replanting OFF (harvest/water only)')

    def sellOn(self):
        self.sellAll = True
        base.localAvatar.setSystemMessage(0, 'Garden Maxer: Auto-sell ON (cashes in the whole basket)')
        self.autoerGui.display.newLine('Auto-sell ON')

    def sellOff(self):
        self.sellAll = False
        base.localAvatar.setSystemMessage(0, 'Garden Maxer: Auto-sell OFF (keep flowers in basket)')
        self.autoerGui.display.newLine('Auto-sell OFF')

    def setStop(self):
        self.shouldContinue = False
        if self.loopTask is not None:
            try:
                self.loopTask.pause()
            except:
                pass
            self.loopTask = None
        self.autoerGui.stopTimer()
        self.autoerGui.display.newLine('Stopped')

    def revertFunctions(self):
        self.removeSafety()
        self.autoerGui.destroy()

gardenMaxer = GardenMaxer()
